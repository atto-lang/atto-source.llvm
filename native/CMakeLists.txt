cmake_minimum_required(VERSION 3.20)
project(atto_native_actor_fault_boundary LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(atto_actor_process_isolation STATIC src/actor_process_isolation.c)
target_include_directories(atto_actor_process_isolation PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(atto_actor_process_isolation PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
find_package(Threads REQUIRED)
add_executable(atto_actor_process_isolation_test tests/actor_process_isolation_test.c)
target_link_libraries(atto_actor_process_isolation_test PRIVATE atto_actor_process_isolation Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(atto_actor_process_isolation_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME process_isolated_actor_fault COMMAND atto_actor_process_isolation_test)
set_tests_properties(process_isolated_actor_fault PROPERTIES TIMEOUT 15)

# Trusted Linux host-side scoped filesystem broker. This is not an Atto guest
# or secure TenantVM runtime. openat2 is deliberately NOT emulated on other OSes.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(atto_host_fs_broker STATIC src/host_fs_broker.c)
  target_include_directories(atto_host_fs_broker PUBLIC include)
  target_link_libraries(atto_host_fs_broker PUBLIC Threads::Threads)
  add_executable(atto_host_fs_broker_test tests/host_fs_broker_test.c)
  target_link_libraries(atto_host_fs_broker_test PRIVATE atto_host_fs_broker)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(atto_host_fs_broker PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_compile_options(atto_host_fs_broker_test PRIVATE -Wall -Wextra -Wconversion -Werror -UNDEBUG)
  endif()
  add_test(NAME linux_scoped_host_fs_broker COMMAND atto_host_fs_broker_test)
  set_tests_properties(linux_scoped_host_fs_broker PROPERTIES TIMEOUT 15)
endif()

# Ed25519 artifact-ceiling verifier: host-only, independent of guest ABI v4.
# Signing keys never live in this library; keypair generation is test-only.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  add_library(atto_host_fs_signed_ceiling STATIC src/host_fs_signed_ceiling.c)

  add_executable(atto_host_fs_signed_ceiling_test tests/host_fs_signed_ceiling_test.c)
  target_link_libraries(atto_host_fs_signed_ceiling_test PRIVATE atto_host_fs_signed_ceiling)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")

# Bounded host I/O worker with one-shot actor-owned tickets. The unit test
# intentionally substitutes an in-memory mock for the rooted broker; a mock
# proves queue/cancellation behavior, NOT filesystem isolation.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(atto_host_fs_async STATIC src/host_fs_async.c)
  target_include_directories(atto_host_fs_async PUBLIC include)
  target_link_libraries(atto_host_fs_async PUBLIC atto_host_fs_broker Threads::Threads)
  add_executable(atto_host_fs_async_mock_test src/host_fs_async.c tests/host_fs_async_test.c)
  target_include_directories(atto_host_fs_async_mock_test PRIVATE include)
  target_link_libraries(atto_host_fs_async_mock_test PRIVATE Threads::Threads)
  add_test(NAME linux_async_host_fs_queue_mock COMMAND atto_host_fs_async_mock_test)
  set_tests_properties(linux_async_host_fs_queue_mock PROPERTIES TIMEOUT 15)

  add_executable(atto_host_fs_io_v5_layout_test tests/host_fs_io_v5_layout_test.c)
  target_include_directories(atto_host_fs_io_v5_layout_test PRIVATE ../src/main/cpp)
  add_test(NAME linux_host_fs_io_v5_layout COMMAND atto_host_fs_io_v5_layout_test)
  set_tests_properties(linux_host_fs_io_v5_layout PROPERTIES TIMEOUT 15)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(atto_host_fs_async PRIVATE -Wall -Wextra -Wconversion -Werror -pedantic)
    target_compile_options(atto_host_fs_async_mock_test PRIVATE -Wall -Wextra -Wconversion -Werror -pedantic -UNDEBUG)
    target_compile_options(atto_host_fs_io_v5_layout_test PRIVATE -Wall -Wextra -Wconversion -Werror -pedantic -UNDEBUG)
  endif()
endif()
